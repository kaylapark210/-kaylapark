Panel builder

build_panel <- function(dat, zip_col, date_col, start, end, label) {
  counts <- dat %>%
    transmute(ZIPCODE = suppressWarnings(as.numeric(.data[[zip_col]])),
              month   = floor_date(.data[[date_col]], "month")) %>%
    filter(ZIPCODE %in% geo$ZIPCODE, month >= start, month <= end) %>%
    count(ZIPCODE, month, name = "y")

  expand_grid(ZIPCODE = geo$ZIPCODE,
              month   = seq(start, end, by = "month")) %>%
    left_join(counts, by = c("ZIPCODE","month")) %>%
    mutate(y = replace_na(y, 0)) %>%
    left_join(geo, by = "ZIPCODE") %>%
    mutate(month_fe  = factor(month),
           rel_month = interval(ANCHOR, month) %/% months(1),
           post      = as.integer(month >  PHASEIN_END),
           phasein   = as.integer(month >= ANCHOR & month <= PHASEIN_END),
           outcome   = label)
}

drop_covid <- function(p) p %>% filter(!(month >= COVID_START & month < COVID_END))

fit_did <- function(p)
  fepois(y ~ treated:post + treated:phasein | ZIPCODE + month_fe,
         data = p, offset = ~log(Population), cluster = ~ZIPCODE)

pct_row <- function(m, lab) {
  b <- coef(m)[["treated:post"]]; s <- se(m)[["treated:post"]]
  tibble(outcome = lab, coef = b, se = s, p = pvalue(m)[["treated:post"]],
         rate_ratio = exp(b), pct = 100*(exp(b)-1),
         lo = 100*(exp(b-1.96*s)-1), hi = 100*(exp(b+1.96*s)-1))
}

pre_window <- function(m, lo, hi) {
  tm  <- grep("^rel_month::", names(coef(m)), value = TRUE)
  r   <- as.integer(sub(".*rel_month::(-?[0-9]+):.*", "\\1", tm))
  sel <- sort(r[r >= lo & r <= hi])
  if (!length(sel)) return(NULL)
  wald(m, keep = paste0("rel_month::(", paste(sel, collapse = "|"), "):"))
}

boot_window <- function(p, lo, hi) {
  lin <- feols(I(y/Population*1e4) ~ i(rel_month, treated, ref = -1) | ZIPCODE + month_fe,
               data = p)
  tm  <- grep("^rel_month::", names(coef(lin)), value = TRUE)
  r   <- as.integer(sub(".*rel_month::(-?[0-9]+):.*", "\\1", tm))
  sel <- tm[r >= lo & r <= hi]
  if (!length(sel)) return(NULL)
  Wald_test(lin, constraints = constrain_zero(sel), vcov = "CR2",
            cluster = p$ZIPCODE, test = "HTZ")
}

try_window <- function(p, lo, hi, lab) {
  out <- tryCatch(boot_window(p, lo, hi), error = function(e) NULL)
  cat("\n", lab, ": ")
  if (is.null(out)) cat("not computable (rank-deficient contrast)\n") else print(out)
}

ri_p <- function(p, B = 1000) {

                      offset = ~log(Population)))[["treated:post"]]
  cand <- geo$ZIPCODE[geo$dist_mi > 2]

    fh <- sample(cand, 2)
    fn <- unique(unlist(lapply(fh, near_of)))
    dd <- p %>% mutate(tp = as.integer(ZIPCODE %in% fn) * post)
    tryCatch(coef(fepois(y ~ tp | ZIPCODE + month_fe, data = dd,
                         offset = ~log(Population)))[["tp"]],
             error = function(e) NA_real_)
  })

         ri_p = mean(abs(draws) >= abs(obs), na.rm = TRUE))
}

honest <- function(m, n_post = 12) {
  ct <- as.data.frame(coeftable(m)); ct$term <- rownames(ct)
  ct <- ct[grepl("^rel_month::", ct$term), ]
  ct$rel <- as.integer(sub(".*rel_month::(-?[0-9]+):.*", "\\1", ct$term))
  keep <- ct[ct$rel %in% c(-5:-2, 0:(n_post-1)), ]

  nm <- grep(pat, names(coef(m)), value = TRUE)
  if (!length(nm)) return(tibble(coef = NA_real_, se = NA_real_, p = NA_real_))

library(ggplot2)
dir.create("figures", showWarnings = FALSE)

save_base <- function(n, expr, w = 9, h = 5) {
  f <- file.path("figures", paste0("Fig ", n))
  pdf(paste0(f, ".pdf"), width = w, height = h)
  expr; dev.off()
  png(paste0(f, ".png"), width = w, height = h, units = "in", res = 600, type = "quartz")
  expr; dev.off()
}

save_gg <- function(n, plot, w = 6, h = 8) {
  f <- file.path("figures", paste0("Fig ", n))
  ggsave(paste0(f, ".pdf"), plot, width = w, height = h)        # uses pdf()
  ggsave(paste0(f, ".png"), plot, width = w, height = h, dpi = 600, bg = "white")
}
